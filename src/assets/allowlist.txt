# every subdomain. Blank lines and lines starting with '#' are ignored. Only
# ports 80 and 443 are proxied; everything else is refused regardless of host.

# This file is the whole allowlist. Nothing else gets out: the firewall drops
# every packet the agent sends directly, so a host that is not named here is
# unreachable even by IP address.
